Hello world ¶

a tiny Ramos program: a Main entrypoint that does some arithmetic, maps a list through a private helper, and prints the results.

module Main
  fn main()
    total = (2 + 3) * 4 - 6 / 2
    doubled = double_all([1, 2, 3, 4])
    println("total is #{total}")
    println("doubled is #{doubled}")

  fnp double_all(list)
    case list
      [] -> []
      [head | tail] -> [head * 2] ++ double_all(tail)

Structs ¶

structs end to end: attributes and methods, a trait with a required function, struct-shaped error results destructured in case, and Struct's introspection functions. Run with ramos run examples/structs/main.rmo.

account.rmo

a struct with attributes, methods that take self, and a trait it implements. Every update returns a new instance; nothing here mutates.

struct Account
  implements Reportable

  attributes
    owner: nil
    balance: 0

  fn summary(self)
    "#{self.owner}: #{self.balance}"

  fn deposit(self, amount)
    self
    | Struct.put(:balance, self.balance + amount)

  fn withdraw(self, amount)
    cond
      amount <= 0 ->
        (:error, WithdrawalError{code: 400, message: "amount must be positive"})
      amount > self.balance ->
        (:error, WithdrawalError{code: 402, message: "insufficient funds"})
      true ->
        updated =
          self
          | Struct.put(:balance, self.balance - amount)
        (:ok, updated)

reportable.rmo

a trait implemented by anything that can describe itself. summary has no body: it is required, and every implementer must define it.

trait Reportable
  fn summary(self)

withdrawal_error.rmo

a struct used as an error payload: a plain :error symbol has nowhere to carry a code and a message, so Account.withdraw returns one of these instead.

struct WithdrawalError
  attributes
    code: 0
    message: nil

main.rmo

module Main
  fn main()
    account = Account{owner: "Andrew", balance: 100}
    println(account.summary())

    account = account.deposit(50)
    println(account.summary())

    account = attempt_withdrawal(account, 30)
    account = attempt_withdrawal(account, 1000)
    account = attempt_withdrawal(account, -5)

    println("fields: #{Struct.keys(account)}")
    println("as a map: #{Struct.to_map(account)}")
    println("is an Account: #{Struct.is_a(account, Account)}")

  fnp attempt_withdrawal(account, amount)
    case account.withdraw(amount)
      (:ok, updated) ->
        println("withdrew #{amount}, new balance #{updated.balance}")
        updated
      (:error, WithdrawalError{message: message}) ->
        println("declined #{amount}: #{message}")
        account
